## subjectForUser 最终原文（imggen.js:199）

```js
function subjectForUser(u, shot) {
  u = u || {};
  return {
    entityType: 'user',
    entityId: String(u.id || ''),
    name: String((u && (u.name || u.wechatName)) || '我'),
    gender: String(u.gender || ''),
    faceText: faceTextFor(u, shot)
  };
}
```
