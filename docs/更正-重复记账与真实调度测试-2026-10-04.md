## 一、修复：抽出共用调度器

### 1.1 核心代码（`scripts/offline-f2-effect-acceptance.js` **L562–L651**）

```js
function runGroupSchedule(spec) {
  var turns = spec.turns || [];
  var send = spec.send;              /* async (turnIndex, input) → turnRecord */
  var persist = spec.persist;        /* async (state) → void；抛错即停止 */
  var log = spec.log || function () {};
  var stopKind = spec.stopKind || function () { return null; };

  var state = {
    轮次: [], 组已登记: false, 停机原因: null, 停止整跑: false,
    已发送数: 0, 持久化次数: 0
  };

  function registerGroup(meta) {
    if (state.组已登记) return false;      /* ★ 幂等：第二次调用无效 */
    state.组已登记 = true;
    state.登记 = meta;
    return true;
  }

  return (async function run() {
    for (var ti = 0; ti < turns.length; ti++) {
      /* ---- 依赖检查：上一轮没有可信回复 → 本轮不发出 ---- */
      if (ti > 0) {
        var chk = prevTurnTrusted(state.轮次[ti - 1]);
        if (!chk.ok) {
          log('  ⛔ 前一轮未产出可信回复（' + chk.why + '）→ **本轮不发出**');
          for (var ri = ti; ri < turns.length; ri++) {
            state.轮次.push({
              轮: ri + 1, 输入: turns[ri], 已落地: false, 未跑: true,
              未跑原因: '依赖轮被拦截：' + chk.why,
              原始请求结果: null, 归因: null, 出站messages: null,
              原始输出: null, 段落: null, 诊断: null
            });
          }
          state.停机原因 = '依赖轮被拦截（上一轮未产出可信回复）';
          state.拦截轮 = ti + 1;
          registerGroup({ 组: spec.id, 拦截轮: ti + 1 });
          /* ★ 失败轮的登记也要落盘 */
          await persist(state);
          return state;
        }
      }

      /* ---- 发这一轮 ---- */
      var rec;
      try { rec = await send(ti, turns[ti]); }
      catch (e) { /* 不吞，记停机并停止后续发送 */ … return state; }
      state.轮次.push(rec);
      state.已发送数++;

      /* ---- ★ 先持久化，再决定下一轮 ---- */
      try { await persist(state); state.持久化次数++; }
      catch (e) {
        state.停机原因 = '持久化失败：' + String(e && e.message);
        state.停止整跑 = true; state.持久化失败 = true;
        throw e;    /* 向上抛，让调用方看到真实错误 */
      }

      /* ---- 整组/整跑停机的硬条件 ---- */
      var sk = stopKind(rec);
      if (sk) { … registerGroup({ 组: spec.id, 硬停止: sk }); await persist(state); return state; }
    }
    /* 正常跑完：登记一次即可（**不再多调一次 persist**） */
    registerGroup({ 组: spec.id, 正常跑完: true });
    return state;
  })();
}
```

### 1.3 ★ 顺带修掉的第二个真 bug：硬停止与依赖拦截的顺序

**症状**：`hardStopKind` 把 `stage==='timeout'|'network'` 也算硬停止
→ G4-2 超时后调度器**在当轮就 `return`**
→ **依赖检查根本没机会跑** → 第三轮既没被标未跑、也没被记进拦截。

**修法**（`hardStopKind`，**L665–L680**）：

| 条件 | 归属 |
|---|---|
| **403** | ✔ 硬停止（换参数也没用） |
| **采集关联异常**（诊断缺失 / 落地却无 `requestId`） | ✔ 硬停止（数据不可信） |
| **超时 / 网络失败** | ❌ **不再算硬停止** → 归 `prevTurnTrusted` 的**依赖拦截** |

**依据**：超时不等于「整跑该停」，它只说明「这一轮没拿到可信回复」。

## 二、实际持久化

### 2.1 修复前：**只有末尾统一写**（问题）

| 时机 | 动作 | 后果 |
|---|---|---|
| 每轮结束 | **仅内存 `turnRecs.push`** | 无文件 |
| 全部组跑完 | `fs.writeFileSync(file, …)`（唯一一次） | ⚠️ **中途崩溃/断电 → 已跑轮次一个不剩**，而那正是「已花钱」的证据 |

### 2.2 修复后：**每轮落盘**（`makeSnapshotWriter`，L685–L724）

| 时机 | 动作 | 类型 |
|---|---|---|
| 每轮 `send` 返回后 | **立即** `await persist(state)` | 文件写入（快照） |
| 依赖拦截 / 硬停止 / send 抛错 | 同样 `await persist(state)` | 文件写入 |
| 每轮结束 | `state.轮次.push(rec)` | **内存** |
| 全部组跑完 | `fs.writeFileSync(file, …)`（正式样本） | 文件写入 |

**快照特性**：
- 路径：`<OUT_DIR>/逐轮快照/快照-<组号>-<序号>-<时间戳>.json`
- 序号从 001 递增
- 内容含：`已发送数` / `持久化次数` / `组已登记` / **`停机原因`** / `页面报错` / **该组全部轮次（含失败轮与未跑轮）**
- 阶段标注：`逐轮快照（运行中，尚未收尾）`
- ⚠️ **凭据不落盘**（`密钥` 字段从未被复制；快照里只有 `maskKey` 脱敏值）

### 2.3 写入失败**不吞**

```js
try { await persist(state); state.持久化次数++; }
catch (e) {
  state.停机原因 = '持久化失败：' + String(e && e.message);
  state.停止整跑 = true; state.持久化失败 = true;
  throw e;    /* 向上抛，让调用方看到真实错误 */
}
```

**为什么必须停**：落盘失败还继续发 = **付了钱没留下证据**。

### 2.4 同步写 + `fsync`

```js
var fd = fs.openSync(f, 'w');
try { fs.writeSync(fd, JSON.stringify(payload, null, 2), null, 'utf8'); fs.fsyncSync(fd); }
finally { fs.closeSync(fd); }
```
异步 `writeFile` 的回调可能在进程退出时才执行 → **返回成功不等于已落盘**。

### 2.5 ★ 顺序保证（测试锁定）

> **发一轮 → 落一盘 → 再发下一轮**

测试实测顺序：`send0,persist1,send1,persist2,send2,persist3`（严格交替）。
